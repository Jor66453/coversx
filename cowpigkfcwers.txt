local function _is_hub_key(k)
	if type(k) ~= "string" then return false end
	if k == "TAG"
	   or k == "COVERHUB_KEY" or k == "COVERHUB_HWID"
	   or k == "__cvh_shared" or k == "__cvh_hb_last"
	   or k == "__coverhub_hub" or k == "__coverhub_unload" or k == "__coverhub_unload_signal"
	   or k == "MLHubUnload" or k == "__MLAim" then
		return true
	end
	local l = k:lower()
	return l:find("coverhub", 1, true) ~= nil
	    or l:find("cvh",      1, true) ~= nil
	    or l:find("mlhub",    1, true) ~= nil
	    or l:find("__ml",     1, true) ~= nil
end
local _safe_g = setmetatable({}, {
	__index = function(_, k)
		if _is_hub_key(k) then return nil end
		return rawget(_G, k)
	end,
	__newindex = function(_, k, v) rawset(_G, k, v) end,
	__iter = function()
		return function(_, prev)
			local k, v = next(_G, prev)
			while k ~= nil do
				if not _is_hub_key(k) then return k, v end
				k, v = next(_G, k)
			end
		end, nil, nil
	end,
})

local _ac_empty
do
	local ok, b = pcall(_ac_compress, es, _ac_fromstr(""), _ac_zstd, 1)
	if ok then _ac_empty = b end
end
local function _ac_fallback()
	if _ac_empty then return _ac_empty end
	return _ac_compress(es, _ac_fromstr(""), _ac_zstd, 1)
end
local invocation = 0
local function ac_handler(payload)
	invocation += 1
	local n = invocation
	local ok_d, d = pcall(_ac_decompress, es, payload, _ac_zstd)
	if not ok_d or not d then return _ac_fallback() end
	local ok_s, raw_src = pcall(_ac_tostr, d)
	if not ok_s or type(raw_src) ~= "string" then return _ac_fallback() end
	if DEBUG then
		pcall(function()
			if not isfolder("ac_probes") then makefolder("ac_probes") end
			writefile("ac_probes/probe_" .. tostring(n) .. ".bin", raw_src)
		end)
	end
	local out = ""
	local ok_c, loaded
	if _ac_execute then
		ok_c, loaded = pcall(_ac_execute, raw_src, _shield_env)
	else
		ok_c, loaded = pcall(_ac_vLuau, raw_src, _shield_env)
	end
	if ok_c and type(loaded) == "function" then
		local ran_ok, ran_out = pcall(loaded)
		if ran_ok and type(ran_out) == "string" then out = ran_out end
	end
	if DEBUG then
		pcall(function() writefile("ac_probes/probe_" .. tostring(n) .. ".out.txt", out) end)
	end
	local ok_r, response = pcall(_ac_compress, es, _ac_fromstr(out), _ac_zstd, 1)
	if not ok_r then return _ac_fallback() end
	return response
end

local function pick_target()
	local root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
	if not root then return end
	local name = current_ability_name()
	local range = effective_range(name) + 30
	local m
	if uis.TouchEnabled and not uis.MouseEnabled then
		local vs = cam.ViewportSize
		m = Vector2.new(vs.X / 2, vs.Y / 2)
	else
		m = uis:GetMouseLocation()
	end
	local best, best_px = nil, math.huge
	for char in pairs(iter_targets()) do
		if not is_friend(char) and not is_whitelisted(char) then
			local hum = char:FindFirstChildOfClass("Humanoid")
			local a = anchor(char)
			if hum and hum.Health > 0 and a then
				local dist = (a.Position - root.Position).Magnitude
				if dist <= range then
					local sp, on = cam:WorldToViewportPoint(a.Position)
					if on and sp.Z > 0 then
						local dx, dy = sp.X - m.X, sp.Y - m.Y
						local px = (dx * dx + dy * dy) ^ 0.5
						if px < best_px then
							best_px = px
							best = char
						end
					end
				end
			end
		end
	end
	return best
end
local BEHIND_ONLY = {}
local function is_behind_target(char)
	if not char then return false end
	local my_root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
	local their = char:FindFirstChild("HumanoidRootPart")
	           or (char:IsA("Model") and char.PrimaryPart)
	if not (my_root and their) then return false end
	if char:GetAttribute("ragdoll") == true then return true end
	local look = their.CFrame.LookVector
	look = Vector3.new(look.X, 0, look.Z)
	if look.Magnitude < 0.05 then return true end
	local dir = (my_root.Position - their.Position)
	dir = Vector3.new(dir.X, 0, dir.Z)
	if dir.Magnitude < 0.1 then return true end
	return look.Unit:Dot(dir.Unit) < 0.3
end

local patched = setmetatable({}, { __mode = "k" })
local wrap_cb
wrap_cb = function(ts, orig)
	local kind = ts.TargetType
	local wrapper = function(info, victim)
		local cur = aim.Current
		if not ((aim.Enabled or osp.enabled or osp.key_on) and cur and cur.Parent) then
			return orig(info, victim)
		end
		if is_tool_ts(ts, orig, info) then return orig(info, victim) end
		local a = anchor(cur)
		local name = (info and rawget(info, "_name")) or current_ability_name()
		if name and (AIM_EXCLUDED[name] or tool_data[name]) then return orig(info, victim) end
		if name and BEHIND_ONLY[name] and not is_behind_target(cur) then
			return orig(info, victim)
		end
		local range = effective_range(name) + 30
		local root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
		if not a then return orig(info, victim) end
		if root and (a.Position - root.Position).Magnitude > range then
			return orig(info, victim)
		end
		if kind == target_type_enum.AreaCheck or kind == target_type_enum.CylinderCast then
			return { [cur] = true }
		elseif kind == target_type_enum.CastResult then
			return a.Position
		elseif victim ~= nil then
			if victim == cur then return cur end
			return nil
		else
			return cur, nil
		end
	end
	return newcclosure and newcclosure(wrapper) or wrapper
end
local function patch_ts(ts)
	if patched[ts] or type(ts) ~= "table" or not ts.Callback then return end
	if is_tool_ts(ts, ts.Callback, nil) then return end
	patched[ts] = ts.Callback
	ts.Callback = wrap_cb(ts, ts.Callback)
end

		local vts = rawget(ts, "ValidTargets")

		local a = anchor(cur)
		if a then rawset(ts, "ValidTargetPoint", a.Position) end
	else
		rawset(ts, "ValidTarget", cur)
	end
end

do
	local function proj_target_pos()
		if not (aim.Enabled and aim.Current and aim.Current.Parent) then return nil end
		local a = anchor(aim.Current)
		if not a then return nil end
		return a.Position, a
	end
	local function caller_is_engine()
		for i = 2, 8 do
			local src = debug.info(i, "s")
			if not src or src == "" then break end
			if src:find("CameraModule", 1, true)
			   or src:find("PlayerModule", 1, true)
			   or src:find("Popper", 1, true)
			   or src:find("ControlModule", 1, true) then
				return true
			end
		end
		return false
	end
	local old_namecall
	old_namecall = hookmetamethod(game, "__namecall", function(self, ...)
		local method = getnamecallmethod()
		if method == "ScreenPointToRay" or method == "ViewportPointToRay" then
			if self == cam and not caller_is_engine() then
				local pos = proj_target_pos()
				if pos then
					local origin = cam.CFrame.Position
					local dir = (pos - origin)
					local len = dir.Magnitude
					if len > 0.001 then
						return Ray.new(origin, dir.Unit)
					end
				end
			end
		elseif method == "Raycast" then
			if self == workspace and not caller_is_engine() then
				local args = { ... }
				local origin, direction = args[1], args[2]
				if typeof(origin) == "Vector3" and typeof(direction) == "Vector3" and direction.Magnitude > 0 then
					local pos, part = proj_target_pos()
					if pos then
						local to_target = pos - origin
						local tt_len = to_target.Magnitude
						if tt_len > 0.001 then
							local tt_unit = to_target.Unit
							local ray_unit = direction.Unit
							if tt_unit:Dot(ray_unit) > 0.985 and direction.Magnitude >= tt_len * 0.9 then
								return {
									Position = pos,
									Instance = part,
									Normal = -ray_unit,

								}
							end

		return old_namecall(self, ...)
	end)

	function Library:Unload()
		for _, c in ipairs(Library._conns) do
			pcall(function() c:Disconnect() end)
		end
		Library._conns = {}
		if Library._gui then
			pcall(function() Library._gui:Destroy() end)
			Library._gui = nil
		end
	end

local function osp_target_already_affected(name, char)
	if not (char and name) then return false end
	local keys = OSP_CC_MAP[name]
	if not keys then return false end
	for _, k in ipairs(keys) do
		if attr_true(char, k) then return true end
		local ok_tag, has = pcall(function() return cs:HasTag(char, k) end)
		if ok_tag and has == true then return true end
	end
	return false
end
local function osp_target_protected(char)
	if not char then return true end
	if attr_true(char, "GodMode") or attr_true(char, "ProtectionMode") then return true end
	local p = plrs:GetPlayerFromCharacter(char)
	if p and (attr_true(p, "GodMode") or attr_true(p, "ProtectionMode")) then return true end
	if attr_true(char, "InPendantTrap") or attr_true(char, "PendantTrap") then return true end
	if attr_true(char, "InAstralProjection") then return true end
	return false
end
local function osp_los_clear(char)
	local me = lp.Character
	if not me then return true end
	local my_pts = {}
	for _, n in ipairs({ "Head", "UpperTorso", "HumanoidRootPart" }) do
		local p = me:FindFirstChild(n)
		if p and p:IsA("BasePart") then my_pts[#my_pts + 1] = p.Position end
	end
	local tgt_pts = {}
	for _, n in ipairs({ "Head", "UpperTorso", "HumanoidRootPart" }) do
		local p = char:FindFirstChild(n)
		if p and p:IsA("BasePart") then tgt_pts[#tgt_pts + 1] = p.Position end
	end
	if #my_pts == 0 or #tgt_pts == 0 then return true end
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { me, char }
	params.IgnoreWater = true
	params.RespectCanCollide = false
	for _, origin in ipairs(my_pts) do
		for _, dest in ipairs(tgt_pts) do
			local dir = dest - origin
			if dir.Magnitude < 2 then return true end
			local ok, result = pcall(workspace.Raycast, workspace, origin, dir, params)
			if ok then
				if not result then return true end
				if result.Instance and result.Instance:IsDescendantOf(char) then return true end
				local mat = result.Material
				if mat == Enum.Material.Grass or mat == Enum.Material.LeafyGrass
				   or mat == Enum.Material.Water or mat == Enum.Material.Glass then
					return true
				end
				local inst = result.Instance
				if inst and (inst.Transparency and inst.Transparency > 0.5) then
					return true
				end
				if inst and (not inst.CanCollide) then return true end
			end
		end
	end
	return false
end
local function ability_accepts(h, char)
	if type(h) ~= "table" or not char then return nil end
	local ts = rawget(h, "_targetSystem")
	local kind = ts and rawget(ts, "TargetType")
	if kind == target_type_enum.AreaCheck or kind == target_type_enum.CylinderCast then
		return nil
	end
	local hdm = rawget(h, "hitDetectionMobile")
	if type(hdm) ~= "function" then
		local mt = getmetatable(h)
		local idx = type(mt) == "table" and mt.__index
		if type(idx) == "table" then hdm = rawget(idx, "hitDetectionMobile") end
	end
	if type(hdm) ~= "function" then return nil end
	local ok, res = pcall(hdm, h, char)
	if ok and res == char then return true end
	return false
end
local function osp_target_char()
	local char = aim.Current
	if not (char and char.Parent) then return nil end
	if is_friend(char) or is_whitelisted(char) then return nil end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local a = anchor(char)
	if not (hum and hum.Health > 0 and a) then return nil end
	local root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
	if not root then return nil end
	local name = current_ability_name()
	local d_cur = name and ability_data[name]
	local range = effective_range(name) + (osp.range_pad or 2)
	if (a.Position - root.Position).Magnitude > range then return nil end
	if not osp_can_affect(char, name, d_cur) then return nil end
	if name and BEHIND_ONLY[name] and not is_behind_target(char) then return nil end
	if osp.smart_gates then
		if osp_target_protected(char) then return nil end
		if name and OSP_SKIP_RAGDOLL[name] and not osp.force_normal[name]
		   and attr_true(char, "ragdoll") then return nil end
		if osp.skip_if_already_affected and osp_target_already_affected(name, char) then return nil end
		if osp.los_check and not osp.force_normal[name] and not osp_los_clear(char) then return nil end
	end
	return char
end
local function osp_can_activate(h, d)
	if real_can_activate then
		local ok, can = pcall(real_can_activate, h._name)
		if ok then return can == true end
	end
	if not d then return true end
	local st, sc = d.statType, d.statCost
	if st and sc and sc > 0 then
		if not player_stats_client then return true end
		local fn = player_stats_client.getStat
		if type(fn) ~= "function" then return true end
		local ok, cur = pcall(fn, st)
		if not ok or type(cur) ~= "number" then return true end
		if cur < sc then return false end
	end
	if lp:GetAttribute("MagicShackles") and d and tostring(d.statType) == "Magic" then return false end
	if lp:GetAttribute("MAGIC_LOCK") and d and tostring(d.statType) == "Magic" then return false end
	return true
end

function osp.fire_pingless(h)
	local rem = osp.RemActivated5
	if not rem then return end
	local name = rawget(h, "_name")
	if type(name) ~= "string" then return end
	local ts = rawget(h, "_targetSystem")
	if not ts then return end
	local gti = ts.GetTargetInfo
	if type(gti) ~= "function" then return end
	local ok, info = pcall(gti, ts)
	if not ok or info == nil then return end
	local info_t = typeof(info)
	if info_t == "table" and next(info) == nil then return end
	local timeout = rawget(h, "_timeout")
	if type(timeout) == "number" and not client_debounce.isAlive(name) then
		pcall(client_debounce.set, name, timeout)
	end
	if info_t == "Vector3" then
		pcall(rem.FireServer, rem, info, http_svc:GenerateGUID(false))
	else
		pcall(rem.FireServer, rem, info)
	end
	local bind = osp.BindActivated
	if bind then pcall(bind.Fire, bind, name) end
end
function osp_once()
	local h = rawget(ability_hdl, "activeAbility")
	if type(h) ~= "table" or type(rawget(h, "_name")) ~= "string" then return end
	local n = h._name
	if AIM_EXCLUDED[n] then return end
	local d = ability_data[n]
	if osp.smart_gates and osp_self_blocked() then return end
	local will_fire_normal = osp.method ~= "pingless" or osp.force_normal[n]
	if will_fire_normal and client_debounce.isAlive(n) then return end
	if osp.skip_projectiles and osp.force_normal[n] then return end
	if not osp_can_activate(h, d) then return end
	local tgt = osp_target_char()
	if not tgt then return end
	local ts = rawget(h, "_targetSystem")
	if ts then
		local kind = rawget(ts, "TargetType")
		if kind == target_type_enum.AreaCheck or kind == target_type_enum.CylinderCast then
			local vts = rawget(ts, "ValidTargets")
			if type(vts) == "table" then vts[tgt] = true end
		elseif kind == target_type_enum.CastResult then
			local a = anchor(tgt)
			if a then rawset(ts, "ValidTargetPoint", a.Position) end
		else
			rawset(ts, "ValidTarget", tgt)
		end
		local has = rawget(ts, "ValidTarget") or rawget(ts, "ValidTargetPoint")
		if not has then
			local vts = rawget(ts, "ValidTargets")
			if type(vts) == "table" and next(vts) then has = true end
		end
		if not has then return end
	end
	local tp_root, tp_orig_cf
	if osp.mini_tp and tgt and not osp.force_normal[n] then
		local my_root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
		local their_root = tgt:FindFirstChild("HumanoidRootPart")
		                or (tgt:IsA("Model") and tgt.PrimaryPart)
		if my_root and their_root then
			tp_orig_cf = my_root.CFrame
			tp_root = my_root
			my_root.CFrame = their_root.CFrame * CFrame.new(0, 0, 3)
		end
	end
	local face_root, face_orig_cf
	if osp.force_normal[n] and tgt then
		local my_root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
		local a = anchor(tgt)
		if my_root and a then
			local flat_target = Vector3.new(a.Position.X, my_root.Position.Y, a.Position.Z)
			if (flat_target - my_root.Position).Magnitude > 0.01 then
				face_orig_cf = my_root.CFrame
				face_root = my_root
				my_root.CFrame = CFrame.lookAt(my_root.Position, flat_target)
			end
		end
	end
	if osp.method == "pingless" and not osp.force_normal[n] then
		osp.fire_pingless(h)
	else
		osp.fire_normal(h)
		if d and d.holding then
			task.wait()
			rawset(h, "_isHolding", false)
			pcall(function() h:deactivate() end)
		end
	end
	if face_root and face_orig_cf then
		face_root.CFrame = face_orig_cf
	end
	if tp_root and tp_orig_cf then
		tp_root.CFrame = tp_orig_cf
	end
end
local osp_last_t = 0
local osp_conn = rsvc.Heartbeat:Connect(function()
	if not osp.enabled then return end
	local cps = osp.cps or 120
	local interval = 1 / cps
	local now = os.clock()
	if osp_last_t == 0 or (now - osp_last_t) > 1 then
		osp_last_t = now - interval
	end
	local max_per_tick = math.max(1, math.ceil(cps / 30))
	local fires = 0
	while (now - osp_last_t) >= interval and fires < max_per_tick do
		pcall(osp_once)
		osp_last_t = osp_last_t + interval
		fires = fires + 1
	end
end)

local function esp_real_health(player, hum)
	if player then
		local tf = workspace:FindFirstChild("PlayerNameTagFolder")
		local tag = tf and tf:FindFirstChild(tostring(player.UserId))
		local bar = tag and tag:FindFirstChild("Health")
		bar = bar and bar:FindFirstChild("Bar")
		bar = bar and bar:FindFirstChild("Progress")
		if bar and bar:IsA("GuiBase2d") then
			local frac = bar.Size.X.Scale
			if type(frac) == "number" and frac >= 0 and frac <= 1 then
				return math.floor(frac * 100 + 0.5)
			end
		end
	end
	if hum then return math.floor(hum.Health) end
	return 0
end

local function esp_update()
	if not esp.Enabled then
		for _, o in pairs(esp_objs) do o.box.Visible = false end
		return
	end
	local camera = workspace.CurrentCamera
	local my_root = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart")
	local alive = {}
	for _, p in ipairs(plrs:GetPlayers()) do if p ~= lp then alive[p] = true end end
	for p in pairs(esp_objs) do if not alive[p] then esp_remove(p) end end
	for p in pairs(alive) do
		local o = esp_objs[p] or esp_make(p)
		local char = p.Character
		local root = char and char:FindFirstChild("HumanoidRootPart")
		local hum  = char and char:FindFirstChildOfClass("Humanoid")
		if not (char and esp_is_spawned(char) and root and hum and hum.Health > 0) then
			o.box.Visible = false
		else
			local sp, on = camera:WorldToViewportPoint(root.Position + Vector3.new(0, 1.5, 0))
			local dist = my_root and (root.Position - my_root.Position).Magnitude or 0
			if not on or sp.Z <= 0 or dist > esp.DistanceLimit then
				o.box.Visible = false
			else
				local slot   = p:GetAttribute("CharacterName") or char:GetAttribute("CharacterName") or ""
				local specie = char:GetAttribute("SpecieType") or ""
				local color  = esp_color(specie, slot)
				local invis  = esp_is_invisible(char)
				local sz     = esp.TextSize
				local t1 = {}
				if esp.ShowSlotName and slot ~= "" then t1[#t1 + 1] = slot end
				if esp.ShowSpecie and specie ~= "" then t1[#t1 + 1] = "(" .. specie .. ")" end
				local s1 = table.concat(t1, " ")
				local t2 = {}
				if esp.ShowDisplayName and p.DisplayName ~= "" then t2[#t2 + 1] = p.DisplayName end
				if esp.ShowUsername and p.Name ~= "" then t2[#t2 + 1] = "(@" .. p.Name .. ")" end
				local s2 = table.concat(t2, " ")
				local t3 = {}
				if esp.ShowHealth then t3[#t3 + 1] = esp_real_health(p, hum) .. "%" end
				if esp.ShowDistance then t3[#t3 + 1] = ('<font color="rgb(231,218,217)">[%d m]</font>'):format(math.floor(dist)) end
				local s3 = table.concat(t3, " ")
				if esp.ShowVisibility then
					local vc = invis and Color3.fromRGB(240, 146, 139) or Color3.fromRGB(136, 234, 141)
					s3 = s3 .. (s3 ~= "" and "  " or "") ..
						('<font color="%s">%s</font>'):format(hex(vc), invis and "Invisible" or "Visible")
				end
				local isfr = esp.ColorFriends and friends.ids[p.UserId] == true
				o.r1.Text, o.r1.TextColor3, o.r1.TextSize, o.r1.Visible = s1, color, sz, s1 ~= ""
				o.r2.Text, o.r2.TextSize, o.r2.Visible = s2, sz, s2 ~= ""
				o.r2.TextColor3 = isfr and esp.FriendsColor or Color3.fromRGB(200, 200, 200)
				o.r3.Text, o.r3.TextSize, o.r3.Visible = s3, sz, s3 ~= ""
				o.box.Position = UDim2.fromOffset(sp.X, sp.Y - 6 + esp.YOffset)
				o.box.Visible = (s1 ~= "" or s2 ~= "" or s3 ~= "")
			end
		end
	end
end

	local function overlapping(a, b)
		if not (a and b and a.AbsolutePosition and b.AbsolutePosition) then return false end

		local bw, bh = b.AbsoluteSize.X, b.AbsoluteSize.Y
		return ax < bx + bw and bx < ax + aw and ay < by + bh and by < ay + ah
	end

local orig_color = setmetatable({}, { __mode = "k" })
local function color_snap(inst)
	if orig_color[inst] ~= nil then return end
	if inst:IsA("Fire") then
		orig_color[inst] = { Color = inst.Color, SecondaryColor = inst.SecondaryColor }
	elseif inst:IsA("Sparkles") then
		orig_color[inst] = { SparkleColor = inst.SparkleColor }
	elseif inst:IsA("Smoke") then
		orig_color[inst] = { Color = inst.Color }
	elseif inst:IsA("BasePart") then
		orig_color[inst] = { Color = inst.Color }
	else
		orig_color[inst] = inst.Color
	end
end
local function color_paint(inst, color)
	color_snap(inst)
	pcall(function()
		if inst:IsA("PointLight") or inst:IsA("SpotLight") or inst:IsA("SurfaceLight") then
			inst.Color = color
		elseif inst:IsA("Fire") then
			inst.Color = color; inst.SecondaryColor = color
		elseif inst:IsA("Sparkles") then
			inst.SparkleColor = color
		elseif inst:IsA("Smoke") then
			inst.Color = color
		elseif inst:IsA("BasePart") then
			if inst.Material == Enum.Material.Neon then inst.Color = color end
		else
			inst.Color = ColorSequence.new(color)
		end
	end)
end
local function color_restore(inst)
	local o = orig_color[inst]
	if o == nil then return end
	pcall(function()
		if type(o) == "table" then
			for k, v in pairs(o) do inst[k] = v end
		else
			inst.Color = o
		end
	end)
	orig_color[inst] = nil
end
local function is_colorable(inst)
	if inst:IsA("ParticleEmitter") or inst:IsA("Trail") or inst:IsA("Beam") then return true end
	if inst:IsA("PointLight") or inst:IsA("SpotLight") or inst:IsA("SurfaceLight") then return true end
	if inst:IsA("Fire") or inst:IsA("Sparkles") or inst:IsA("Smoke") then return true end
	if inst:IsA("BasePart") and inst.Material == Enum.Material.Neon then return true end
	return false
end
local function paint_template_tree(root, on, color)
	if not root then return end
	for _, d in ipairs(root:GetDescendants()) do
		if is_colorable(d) then
			if on then color_paint(d, color) else color_restore(d) end
		end
	end
	if is_colorable(root) then
		if on then color_paint(root, color) else color_restore(root) end
	end
end

	for _, d in ipairs(assets:GetDescendants()) do

	end
end

		end
		wheel_hooks = {}
	end
end

local name_backups = setmetatable({}, { __mode = "k" })
local function mutate_name_table(t, on)
	local function mutate_entry(entry, key, bucket)
		for _, dk in ipairs({ "displayName", "DisplayName" }) do
			local cover = rawget(entry, dk)
			if type(cover) == "string" then
				local orig = type(rawget(entry, "name")) == "string" and entry.name
				          or type(rawget(entry, "Name")) == "string" and entry.Name
				          or (type(key) == "string" and key or nil)
				if orig and cover ~= orig then
					bucket[tostring(key).."|"..dk] = { entry = entry, key = dk, cover = cover }
					entry[dk] = orig
				end
			end
		end
	end
	if on then
		local bucket = name_backups[t] or {}
		name_backups[t] = bucket
		for key, entry in pairs(t) do
			if type(entry) == "table" then
				mutate_entry(entry, key, bucket)
				for inner_key, inner_entry in pairs(entry) do
					if type(inner_entry) == "table" then
						mutate_entry(inner_entry, inner_key, bucket)
					end
				end
			end
		end
	else
		local bucket = name_backups[t]
		if bucket then
			for _, info in pairs(bucket) do
				pcall(function() info.entry[info.key] = info.cover end)
			end
			name_backups[t] = nil
		end
	end
end
local cover_to_real = nil
local real_to_cover = nil
local function rebuild_cover_maps()
	local c2r, r2c = {}, {}
	for _, b in pairs(name_backups) do
		for _, info in pairs(b) do
			local entry = info.entry
			local cover = info.cover
			local real
			if type(entry) == "table" then
				if type(rawget(entry, "name")) == "string" then real = entry.name
				elseif type(rawget(entry, "Name")) == "string" then real = entry.Name end
			end
			if cover and real and cover ~= real then
				c2r[cover] = real
				r2c[real] = cover
			end
		end
	end
	for _, pl in ipairs(plrs:GetPlayers()) do
		local cover = pl:GetAttribute("DisplayName")
		local real  = pl:GetAttribute("CharacterName")
		if type(cover) == "string" and type(real) == "string"
		   and #cover > 0 and #real > 0 and cover ~= real then
			c2r[cover] = real
			r2c[real] = cover
		end
	end
	cover_to_real = c2r
	real_to_cover = r2c
end
local relabel_state = setmetatable({}, { __mode = "k" })
local function relabel(d, map)
	local txt = d.Text
	if type(txt) ~= "string" or #txt < 3 then return end
	if not txt:find("%a") then return end
	if relabel_state[d] and relabel_state[d].set == txt then return end
	local new = map[txt]
	if not new then
		local trimmed = txt:gsub("^%s+", ""):gsub("%s+$", "")
		if trimmed ~= txt then new = map[trimmed] end
	end
	if not new then
		local stripped = txt:gsub("<[^>]->", "")
		if stripped ~= txt and stripped ~= "" then
			local mapped = map[stripped]
			if mapped then new = (txt:gsub(stripped:gsub("([^%w])", "%%%1"), mapped, 1)) end
		end
	end
	if not new then
		for cov, real in pairs(map) do
			if #cov >= 3 and txt:find(cov, 1, true) then
				local replaced = (txt:gsub(cov:gsub("([^%w])", "%%%1"), real, 1))
				if replaced ~= txt then new = replaced; break end
			end
		end
	end
	if new and new ~= txt then
		pcall(function() d.Text = new end)
		relabel_state[d] = { set = new }
	end
end

local sweep_conns = {}
local label_prop_conns = setmetatable({}, { __mode = "k" })
local function text_matches_cover(txt)
	if type(txt) ~= "string" or not cover_to_real then return false end
	if cover_to_real[txt] then return true end
	local trimmed = txt:gsub("^%s+", ""):gsub("%s+$", "")
	if trimmed ~= txt and cover_to_real[trimmed] then return true end
	local stripped = txt:gsub("<[^>]->", "")
	if stripped ~= txt and cover_to_real[stripped] then return true end
	for cov in pairs(cover_to_real) do
		if #cov >= 3 and txt:find(cov, 1, true) then return true end
	end
	return false
end
local function watch_label(d)
	if not cover_to_real then return end
	relabel(d, cover_to_real)
	if label_prop_conns[d] then return end
	if not text_matches_cover(d.Text) and not relabel_state[d] then return end
	label_prop_conns[d] = d:GetPropertyChangedSignal("Text"):Connect(function()
		if cos.RestoreNames and cover_to_real then relabel(d, cover_to_real) end
	end)
end
local function start_text_sweep()
	rebuild_cover_maps()
	if cover_to_real then
		for _, root in ipairs(sweep_roots()) do
			for _, d in ipairs(root:GetDescendants()) do
				if d:IsA("TextLabel") or d:IsA("TextButton") then
					watch_label(d)
				end
			end
		end
	end
	for _, root in ipairs(sweep_roots()) do
		sweep_conns[#sweep_conns+1] = root.DescendantAdded:Connect(function(d)
			if not cos.RestoreNames then return end
			if d:IsA("TextLabel") or d:IsA("TextButton") then
				task.defer(watch_label, d)
			end
		end)
	end
	task.spawn(function()
		while cos.RestoreNames and cover_to_real do
			task.wait(2)
			if not (cos.RestoreNames and cover_to_real) then return end
			rebuild_cover_maps()
			for _, root in ipairs(sweep_roots()) do
				for _, d in ipairs(root:GetDescendants()) do
					if d:IsA("TextLabel") or d:IsA("TextButton") then
						relabel(d, cover_to_real)
					end
				end
			end
		end
	end)
end
local function stop_text_sweep()
	for _, c in ipairs(sweep_conns) do pcall(function() c:Disconnect() end) end
	sweep_conns = {}
	for d, c in pairs(label_prop_conns) do
		pcall(function() c:Disconnect() end)
		label_prop_conns[d] = nil
	end
	if real_to_cover then
		for _, root in ipairs(sweep_roots()) do
			for _, d in ipairs(root:GetDescendants()) do
				if d:IsA("TextLabel") or d:IsA("TextButton") then
					relabel(d, real_to_cover)
				end
			end
		end
	end
	cover_to_real = nil
	real_to_cover = nil
end
local function sync_one_humanoid(pl, on)
	local char = pl.Character
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	if not hum then return end
	local real = (char and char:GetAttribute("CharacterName")) or pl:GetAttribute("CharacterName")
	if on and real then
		if hum:GetAttribute("__coverhub_orig_disp") == nil then
			hum:SetAttribute("__coverhub_orig_disp", hum.DisplayName)
		end
		pcall(function() hum.DisplayName = real end)
	elseif not on then
		local orig = hum:GetAttribute("__coverhub_orig_disp")
		if orig then pcall(function() hum.DisplayName = orig end) end
		hum:SetAttribute("__coverhub_orig_disp", nil)
	end
end
local function sync_humanoid_names(on)
	for _, pl in ipairs(plrs:GetPlayers()) do sync_one_humanoid(pl, on) end
end
local function apply_restore_names(on)
	for _, parts in ipairs(name_module_paths) do
		local mod = resolve_data_module(parts)
		if mod then
			local ok, tbl = pcall(require, mod)
			if ok and type(tbl) == "table" then mutate_name_table(tbl, on) end
		end
	end
	sync_humanoid_names(on)
	if on then start_text_sweep() else stop_text_sweep() end
end

local fps = { enabled = false }

local fps_restore = {}
local function fps_snap(prop_target, key)
	local saved = {}
	for _, inst in ipairs(prop_target) do
		local ok, v = pcall(function() return inst[key] end)
		if ok then saved[inst] = v end
	end
	return saved
end
local function fps_strip_lighting(on)
	local lighting = game:GetService("Lighting")
	if on then
		if fps_restore.lighting_fx == nil then
			fps_restore.lighting_fx = {}
			for _, d in ipairs(lighting:GetDescendants()) do
				if d:IsA("BlurEffect") or d:IsA("SunRaysEffect")
				   or d:IsA("DepthOfFieldEffect") then
					local ok, en = pcall(function() return d.Enabled end)
					if ok then
						fps_restore.lighting_fx[d] = en
						pcall(function() d.Enabled = false end)
					end
				end
			end
		end
		if fps_restore.globalshadows == nil then
			fps_restore.globalshadows = lighting.GlobalShadows
			pcall(function() lighting.GlobalShadows = false end)
		end
	else
		if fps_restore.lighting_fx then
			for d, v in pairs(fps_restore.lighting_fx) do
				pcall(function() d.Enabled = v end)
			end
			fps_restore.lighting_fx = nil
		end
		if fps_restore.globalshadows ~= nil then
			pcall(function() lighting.GlobalShadows = fps_restore.globalshadows end)
			fps_restore.globalshadows = nil
		end
	end
end
local function fps_strip_terrain(on)
	local t = workspace:FindFirstChildOfClass("Terrain")
	if not t then return end
	if on then
		if fps_restore.terrain == nil then
			fps_restore.terrain = {
				WaterWaveSize = t.WaterWaveSize,
				WaterWaveSpeed = t.WaterWaveSpeed,
				WaterReflectance = t.WaterReflectance,
				WaterTransparency = t.WaterTransparency,
			}
		end
		pcall(function()
			t.WaterWaveSize = 0
			t.WaterWaveSpeed = 0
			t.WaterReflectance = 0
			t.WaterTransparency = 1
		end)
	elseif fps_restore.terrain then
		for k, v in pairs(fps_restore.terrain) do
			pcall(function() t[k] = v end)
		end
		fps_restore.terrain = nil
	end
end
local function fps_quality(on)
	local ok, ugs = pcall(function() return settings():GetService("UserGameSettings") end)
	if not ok or not ugs then return end
	if on then
		if fps_restore.quality == nil then
			fps_restore.quality = ugs.SavedQualityLevel
		end
		pcall(function() ugs.SavedQualityLevel = Enum.SavedQualitySetting.QualityLevel3 end)
	elseif fps_restore.quality ~= nil then
		pcall(function() ugs.SavedQualityLevel = fps_restore.quality end)
		fps_restore.quality = nil
	end
end
local function fps_apply(on)
	fps.enabled = on
	fps_strip_lighting(on)
	fps_strip_terrain(on)
	fps_quality(on)
end
